{{- /* Classement ? Renvoie un booléen, qui commande ensemble la mention « sponsorisée » et le disclaimer de classement. Ordre : champ `classement` du front matter (true ou false, prioritaire) ; sinon title, h1 ou seoTitle qui correspond au motif de classement de la langue ; sinon liste de marques classées (`items`). Articles seulement. */ -}} {{- $p := . -}} {{- $r := false -}} {{- if partial "pbn/is-article.html" $p -}} {{- if isset $p.Params "classement" -}} {{- $r = eq (printf "%v" $p.Params.classement) "true" -}} {{- else -}} {{- $base := `meilleur|meilleure|top|classement|comparatif|compar[ée]s?|alternatives|best|ranking|comparison|compared|\bvs\b` -}} {{- /* « quel… choisir » n'est pas un motif : il attrapait des guides sans marque (decision de Damien, mention sur les seuls classements de marques). Un « quel X choisir » qui classe des marques porte `classement: true`. */ -}} {{- $re := printf `(?i)(%s)` $base -}} {{- if in (slice "es" "it" "de") $p.Language.Lang -}} {{- $re = printf `(?i)(%s|mejores|comparativa|alternativas|migliori|classifica|confronto|alternative|beste|vergleich|test)` $base -}} {{- end -}} {{- $txt := delimit (slice $p.Title ($p.Params.h1 | default "") ($p.Params.seoTitle | default $p.Params.seo_title | default "")) " | " -}} {{- $r = or (gt (len (findRE $re $txt 1)) 0) (gt (len ($p.Params.items | default slice)) 0) -}} {{- end -}} {{- end -}} {{- return $r -}}
